<script>
 const questions=[
 {q:"48 ÷ 6 + 7 = ?",a:["8","11","13","15"],correct:3},
 {q:"9 × 7 − 13 = ?",a:["50","49","54","63"],correct:0},
 {q:"(36 + 24) ÷ 6 = ?",a:["8","9","10","12"],correct:2},
 {q:"5² + 6 = ?",a:["25","30","31","36"],correct:2},
 {q:"144 ÷ 12 + 8 = ?",a:["18","20","22","24"],correct:1},
 {q:"3/4 + 1/4 = ?",a:["1/2","1","3/8","2"],correct:1},
 {q:"15 × 4 ÷ 6 = ?",a:["8","10","12","15"],correct:1},
 {q:"100 − 8 × 7 = ?",a:["44","36","56","24"],correct:0},
 {q:"2³ × 3 = ?",a:["12","18","24","36"],correct:2},
 {q:"√81 + 6 = ?",a:["12","13","15","18"],correct:2}
];

const $=id=>document.getElementById(id);

let qi=0,coins=12,lives=3,stage=3,distance=320,lane=0,lessonIndex=0;
let fuel=100,fuelTimer=null,modalIndex=0,correctThisStop=0,answeredModal=false;
let starsCaught=0;

function showPanel(which){
 document.querySelectorAll('.panel').forEach(p=>p.classList.remove('active'));
 $(which).classList.add('active');

 if(which==='race'){
  startRace();
 }else{
  stopRace();
 }

 document.querySelector('.games').scrollIntoView({
  behavior:'smooth',block:'start'
 });
}

function startRace(){
 if(fuelTimer || $('questionModal').classList.contains('show'))return;

 fuelTimer=setInterval(()=>{
  if(!$('race').classList.contains('active')){
   stopRace();
   return;
  }

  fuel=Math.max(0,fuel-2);
  updateHud();

  if(fuel<=0){
   stopRace();
   openFuelQuestions();
  }
 },1000);
}

function stopRace(){
 if(fuelTimer){
  clearInterval(fuelTimer);
  fuelTimer=null;
 }
}

function updateHud(){
 $('coins').textContent=coins;
 $('lives').textContent=lives;
 $('stage').textContent=stage+'/10';
 $('distance').textContent=distance+' m';
 $('progress').style.width=(stage*10)+'%';
 $('fuelText').textContent=fuel+'%';
 $('fuelFill').style.width=fuel+'%';
}

function move(dir){
 if(!$('race').classList.contains('active'))return;
 if($('questionModal').classList.contains('show'))return;

 lane=Math.max(-1,Math.min(1,lane+dir));
 $('car').style.left=(50+lane*20)+'%';

 distance+=1;
 $('distance').textContent=distance+' m';

 // Mashina harakatlanganda yulduz yig‘ish
 if(Math.random()<0.45){
  coins++;
  starsCaught++;
  $('starCount').textContent=Number($('starCount').textContent)+1;
  $('coins').textContent=coins;
 }
}

$('leftBtn').addEventListener('click',()=>move(-1));
$('rightBtn').addEventListener('click',()=>move(1));

let touchX=null;

$('scene').addEventListener('touchstart',e=>{
 touchX=e.touches[0].clientX;
},{passive:true});

$('scene').addEventListener('touchend',e=>{
 if(touchX===null)return;

 let dx=e.changedTouches[0].clientX-touchX;

 if(Math.abs(dx)>25){
  move(dx>0?1:-1);
 }

 touchX=null;
},{passive:true});

document.addEventListener('keydown',e=>{
 if(e.key==='ArrowLeft')move(-1);
 if(e.key==='ArrowRight')move(1);
});

function openFuelQuestions(){
 $('fuelstation').classList.add('show');
 $('raceNotice').textContent='⛽ Yoqilg‘i tugadi — savollarga javob bering!';

 correctThisStop=0;
 modalIndex=0;

 setTimeout(()=>{
  $('questionModal').classList.add('show');
  renderModalQuestion();
 },700);
}

function renderModalQuestion(){
 const q=questions[(qi+modalIndex)%questions.length];

 $('questionCount').textContent=(modalIndex+1)+' / 3-savol';
 $('questionText').textContent=q.q;
 $('answers').innerHTML='';
 $('modalMessage').textContent='';

 answeredModal=false;

 q.a.forEach((v,i)=>{
  const b=document.createElement('button');
  b.className='choice';
  b.textContent=String.fromCharCode(65+i)+') '+v;
  b.onclick=()=>answerModal(i,q.correct,b);
  $('answers').appendChild(b);
 });
}function answerModal(index,correct,btn){
 if(answeredModal)return;

 answeredModal=true;

 [...$('answers').children].forEach((b,i)=>{
  b.disabled=true;
  if(i===correct)b.classList.add('correct');
 });

 if(index===correct){
  correctThisStop++;
  $('modalMessage').textContent='To‘g‘ri javob! ⭐';
  $('modalMessage').style.color='#07834b';
 }else{
  $('modalMessage').textContent='Bu safar xato. Keyingi savolga o‘ting.';
  $('modalMessage').style.color='#bd3b30';
  btn.classList.add('wrong');
 }

 setTimeout(()=>{
  modalIndex++;

  if(modalIndex<3){
   renderModalQuestion();
  }else{
   finishFuelStop();
  }
 },850);
}

function finishFuelStop(){
 const reward=
  correctThisStop===1?3:
  correctThisStop===2?6:
  correctThisStop===3?10:0;

 distance+=reward;
 stage=Math.min(10,stage+1);
 fuel=100;

 $('questionModal').classList.remove('show');
 $('fuelstation').classList.remove('show');

 $('raceNotice').textContent=
  correctThisStop===0
  ?'Bu safar yoqilg‘i bonusi yo‘q. Keyingi safar urinib ko‘ring!'
  :'Ajoyib! '+correctThisStop+
   ' ta to‘g‘ri javob — '+reward+' m oldinga! ⛽';

 $('message').textContent='Yangi yo‘l boshlandi — yulduzchalarni yig‘ing!';

 qi=(qi+3)%questions.length;
 updateHud();
 startRace();
}

function resetGame(){
 stopRace();

 qi=0;
 coins=12;
 lives=3;
 stage=3;
 distance=320;
 lane=0;
 fuel=100;
 starsCaught=0;

 $('car').style.left='50%';
 $('starCount').textContent='12';

function newLessonQuestion(){
 lessonIndex++;
 renderLesson();
}

function renderLesson(){
 const qs=[
  {q:'36 ÷ 4 + 6 = ?',a:['9','12','15','18'],c:2},
  {q:'7 × 8 = ?',a:['54','56','64','48'],c:1},
  {q:'25% of 80 = ?',a:['10','15','20','25'],c:2}
 ];

 const q=qs[lessonIndex%qs.length];

 $('lessonQuestion').textContent=q.q;
 $('lessonAnswers').innerHTML='';
 $('lessonMessage').textContent='';

 q.a.forEach((v,i)=>{
  const b=document.createElement('button');
  b.className='choice';
  b.textContent=String.fromCharCode(65+i)+') '+v;

  b.onclick=()=>{
   [...$('lessonAnswers').children].forEach(x=>x.disabled=true);

   if(i===q.c){
    b.classList.add('correct');
    $('lessonMessage').textContent='Barakalla! To‘g‘ri javob! ⭐';
   }else{
    $('lessonAnswers').children[q.c].classList.add('correct');
    b.classList.add('wrong');
    $('lessonMessage').textContent=
     'Yaxshi urinish! To‘g‘ri javob yashil rangda.';
   }
  };

  $('lessonAnswers').appendChild(b);
 });
}

renderLesson();
updateHud();
</script>
